import hmac, hashlib

def verify_webhook(secret, event_id, timestamp, raw_body, signature):
    # signature is formatted as v1=<hex>
    msg = f"{event_id}.{timestamp}.{raw_body}".encode()
    expected = "v1=" + hmac.new(secret.encode(), msg, hashlib.sha256).hexdigest()
    return hmac.compare_digest(expected, signature)
